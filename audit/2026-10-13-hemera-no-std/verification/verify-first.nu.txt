def payload [lookup: record, root: path, name: string] {
    open --raw ($root | path join objects ($lookup | get $name)) | into binary
}
def json [lookup: record, root: path, name: string] {
    payload $lookup $root $name | decode utf-8 | from json
}
def checked-tar [archive: path, dest: path, compressed: bool] {
    let names = if $compressed { ^tar -tzf $archive | complete } else { ^tar -tf $archive | complete }
    let types = if $compressed { ^tar -tvzf $archive | complete } else { ^tar -tvf $archive | complete }
    if $names.exit_code != 0 or $types.exit_code != 0 or ($names.stdout | lines | any {|p| $p =~ '(^/|(^|/)\.\.(/|$))'}) or ($types.stdout | lines | any {|p| not ($p =~ '^[-d]')}) { error make {msg: 'unsafe or unreadable tar'} }
    mkdir $dest
    let result = if $compressed { ^tar -xzf $archive -C $dest | complete } else { ^tar -xf $archive -C $dest | complete }
    if $result.exit_code != 0 { error make {msg: $result.stderr} }
    $names.stdout | lines | where {|p| not ($p | str ends-with '/')}
}
def command-record [records: record, n: int] { $records | get ($n | into string) }
def stream [lookup: record, dest: path, records: record, n: int, which: string] {
    let r = (command-record $records $n)
    payload $lookup $dest $'code/support/execution/records/($r.record)/($which)' | decode utf-8
}

    let base = ($audit | path join $c.base.relative_path | path expand)
    for pair in [[evidence.tar.gz $c.archive_sha256] [index.json $c.index_sha256]] {

    }

    let base_result = ($base_run.stdout | from json)

    let lookup = ($index | select path sha256 | transpose --ignore-titles --header-row --as-record)
    let sizes = ($index | select path bytes | transpose --ignore-titles --header-row --as-record)
    let groups = ($index | group-by sha256)
    let made = (^mktemp -d /tmp/nox-no-std-execution-verify.XXXXXX | complete)

    let dest = ($made.stdout | str trim)
    try {
        let bnames = (checked-tar ($base | path join evidence.tar.gz) $dest true)
        if ($bnames | sort) != ($base_objects | each {|h| $'objects/($h)'}) { error make {msg: 'base archive object set'} }
        let lnames = (checked-tar ($audit | path join evidence.tar.gz) $dest true)
        if ($lnames | sort) != ($local_objects | each {|h| $'objects/($h)'}) { error make {msg: 'local archive object set'} }
        for h in $objects {
            let p = ($dest | path join objects $h)
            let b = (open --raw $p | into binary)
            if ($p | path type) != file or ($b | hash sha256) != $h or ($groups | get $h | any {|r| $r.bytes != ($b | bytes length)}) { error make {msg: 'exact object bytes'} }
        }
        let m = (json $lookup $dest code/manifest.json)
        let support = (json $lookup $dest code/support-index.json)
        if ($support | length) != $m.support_files or ($support.file | uniq | length) != $m.support_files { error make {msg: 'support count'} }
        for r in $support {
            if ($lookup | get $'code/($r.file)') != $r.sha256 or ($sizes | get $'code/($r.file)') != $r.bytes { error make {msg: 'support identity'} }
        }
        let sections = (json $lookup $dest code/sections.json)
        let parts = ($sections | enumerate | each {|v|
            let s = $v.item
            let b = (payload $lookup $dest $'code/($s.file)')
            if $s.number != ($v.index + 1) or ($b | bytes length) != $s.bytes or ($b | hash sha256) != $s.sha256 { error make {msg: 'section identity'} }
            let head = $"\n===== ($s.number): ($s.title) =====\nfile: ($s.file)\nbytes: ($s.bytes); sha256: ($s.sha256)\n"
            let extra = if 'source' in ($s | columns) {
                let full = (payload $lookup $dest $'code/($s.source)')
                if ($full | hash sha256) != $s.source_sha256 or ($full | bytes at $s.byte_offset..($s.byte_offset + $s.byte_length - 1)) != $b { error make {msg: 'exact excerpt'} }
                $"excerpt of ($s.source) byte offset ($s.byte_offset), length ($s.byte_length); full sha256: ($s.source_sha256)\n"
            } else { '' }
            [($head | into binary) ($extra | into binary) ("----- BEGIN -----\n" | into binary) $b ("\n----- END -----\n" | into binary)] | bytes collect
        } | bytes collect)
        let review = [("Nox Hemera no_std feature boundary — frozen CODE v1\n" | into binary) $parts] | bytes collect

        if (json $lookup $dest code/omissions.json | get files) != ($support | where {|r| not ($r.file in $sections.file)}) { error make {msg: 'exact reviewer omissions'} }
        let plan_support = (json $lookup $dest code/support/plan/support-index.json)
        let base_map = ($base_index | select path sha256 | transpose --ignore-titles --header-row --as-record)
        for r in $plan_support {

        }
        let record_map = ($records | each {|r| {name: ($r.record | split row '-' | first), value: $r}} | transpose --ignore-titles --header-row --as-record)

                } else { $b }

            }
        }
        if $source_count != 536 { error make {msg: 'owning source closure'} }
        for pair in [['rs/Cargo.toml.txt' proposed-Cargo.toml.txt] [CLAUDE.md proposed-CLAUDE.md.txt]] {

        if not $after.all_source_before_after_equal or not $after.current_worktree_diff_equals_freeze or $after.patch_sha256 != $freeze.patch_sha256 { error make {msg: 'recorded gate-source drift check'} }
        let pi = (json $lookup $dest code/support/execution/candidate-package-inputs.json)
        let archive = ($dest | path join objects ($lookup | get code/support/execution/candidate-package.crate))

        if ($package_paths | sort) != ($pi.entries | each {|r| $'cyber-nox-0.2.0/($r.file)'} | sort) { error make {msg: 'normalized package file set'} }
        for r in $pi.entries {
